# ============================================================

# Copy this file to ".env" (remove the .txt extension) and fill in your values.
# The .env file is NOT committed to git (it contains secrets).

# ── LLM Provider ────────────────────────────────────────────
# Choose one: groq (free), gemini, grok, openai
LLM_PROVIDER=groq

# Groq API — FREE tier, fastest (recommended)
# Get key at: https://console.groq.com/keys
GROQ_API_KEY=your_groq_api_key_here
# GROQ_MODEL=llama-3.1-8b-instant    # fast
# GROQ_MODEL=llama-3.1-70b-versatile # better quality

# Google Gemini API — free tier
# Get key at: https://aistudio.google.com/apikey
GOOGLE_API_KEY=your_google_api_key_here
# GEMINI_MODEL=gemini-2.0-flash

# Grok API (X.AI) — requires credits
# Get key at: https://console.x.ai/
GROK_API_KEY=your_grok_api_key_here

# OpenAI API — requires credits
# Get key at: https://platform.openai.com/api-keys
OPENAI_API_KEY=your_openai_api_key_here

# ── Email Notifications ──────────────────────────────────────
# Sends email to IT engineer when a new ticket is assigned
SMTP_SERVER=smtp.gmail.com
SMTP_PORT=587
SMTP_USERNAME=your-gmail@gmail.com
SMTP_PASSWORD=your-16-char-app-password
NOTIFICATION_EMAIL=subrahmanyam.pillalamarri@cognida.ai

# Gmail App Password setup:
# 1. Go to https://myaccount.google.com/apppasswords
# 2. Generate App Password for "Mail"
# 3. Use that 16-character password as SMTP_PASSWORD
# Note: Regular Gmail password will NOT work

# ── PostgreSQL Database ──────────────────────────────────────
# Set USE_DATABASE=true for persistent tickets + chat history
# Run: python scripts/setup_db.py  (after creating the database)
USE_DATABASE=false
DB_HOST=localhost
DB_PORT=5432
DB_NAME=helpdesk_db
DB_USER=postgres
DB_PASSWORD=your_postgres_password

# ── JWT Authentication ───────────────────────────────────────
JWT_SECRET_KEY=cognida-helpdesk-secret-key-2026
JWT_ALGORITHM=HS256
JWT_EXPIRE_HOURS=24
